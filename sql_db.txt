subj table

CREATE TABLE public.subj (
	s_id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
	s_name varchar(32) COLLATE "ru-RU-x-icu" NOT NULL,
	s_type int GENERATED ALWAYS AS IDENTITY NOT NULL,
	amount double precision NOT NULL,
	currency varchar(3) NOT NULL,
	operation_date date NOT NULL,
	description varchar(32) COLLATE "ru-RU-x-icu" NULL,
	tags text COLLATE "ru-RU-x-icu" NULL,
	created_at timestamp NOT NULL,
	created_by varchar(64) NOT NULL,
	updated_at timestamp NOT NULL,
	updated_by varchar(64) NOT NULL,
	check_account_id bigint NOT NULL,
	periodicity_id int NULL,
	CONSTRAINT subj_id_pk PRIMARY KEY (s_id),
	CONSTRAINT s_type_unique UNIQUE (s_type)
);
COMMENT ON TABLE public.subj IS 'Статья бюджета';

-- Column comments

COMMENT ON COLUMN public.subj.s_id IS 'Уникальный идентификатор статьи бюджета.';
COMMENT ON COLUMN public.subj.s_name IS 'Наименование статьи бюджета.';
COMMENT ON COLUMN public.subj.s_type IS 'Ссылка на справочник типов.';
COMMENT ON COLUMN public.subj.amount IS 'Сумма';
COMMENT ON COLUMN public.subj.currency IS 'Валюта';
COMMENT ON COLUMN public.subj.operation_date IS 'Дата проведения финансовой операции';
COMMENT ON COLUMN public.subj.description IS 'Подробное описание операции';
COMMENT ON COLUMN public.subj.tags IS 'Массив тегов для категоризации';
COMMENT ON COLUMN public.subj.created_at IS 'Дата и время создания записи';
COMMENT ON COLUMN public.subj.created_by IS 'Идентификатор пользователя, который создал запись';
COMMENT ON COLUMN public.subj.updated_at IS 'Дата и время последнего обновления записи';
COMMENT ON COLUMN public.subj.updated_by IS 'Идентификатор пользователя, обновившего запись';
COMMENT ON COLUMN public.subj.check_account_id IS 'Идентификатор счета, к которому относится статья бюджета.';
COMMENT ON COLUMN public.subj.periodicity_id IS 'Ссылка на спрвочник типов переодичности';



subj_category table

check_account

CREATE TABLE public.check_account (
	cheack_account_id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
	account_name varchar(32) COLLATE "ru-RU-x-icu" NOT NULL,
	account_type_id bigint NOT NULL,
	acc_currency varchar(3) NOT NULL,
	acc_balance double precision NOT NULL,
	acc_initial_balance double precision NOT NULL,
	acc_owner_name varchar(64) COLLATE "ru-RU-x-icu" NOT NULL,
	acc_owner_id varchar(64) NULL,
	bank_name varchar(64) COLLATE "ru-RU-x-icu" NULL,
	acc_is_active boolean NOT NULL,
	acc_is_default boolean NOT NULL,
	credit_limit double precision NULL,
	CONSTRAINT check_account_id_uni UNIQUE (cheack_account_id),
	CONSTRAINT acc_type_id_pk PRIMARY KEY (account_type_id)

COMMENT ON COLUMN public.check_account.acc_owner_name IS 'Имя владельца счета';
COMMENT ON COLUMN public.check_account.acc_owner_id IS 'ID владельца счета';
COMMENT ON COLUMN public.check_account.bank_name IS 'Название банка';

periodicity_table

CREATE TABLE public.periodicity (
	periodicity_id int NOT NULL,
	period_name varchar(32) COLLATE "ru-RU-x-icu" NULL,
	period_code varchar(16) NOT NULL,
	days_interval int NULL,
	sort_order int NOT NULL
);
COMMENT ON TABLE public.periodicity IS 'Типы переодичности операций';

-- Column comments

COMMENT ON COLUMN public.periodicity.periodicity_id IS 'Внешняя ссылка на справочник типов периодичности';
COMMENT ON COLUMN public.periodicity.period_name IS 'Название периода - "Ежемесячно"';
COMMENT ON COLUMN public.periodicity.period_code IS 'Системное название - monthly';
COMMENT ON COLUMN public.periodicity.days_interval IS 'Интервал в днях';
COMMENT ON COLUMN public.periodicity.sort_order IS 'Порядок в выпадающем списке';
